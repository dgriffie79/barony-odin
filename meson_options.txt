# Dead ends - NOT to be ported. The Odin port does not target Steam/Epic/PlayFab
# online services (see README). The options are kept only so the CMake-era
# option surface is preserved; enabling them has no effect on a working build
# (they still need the proprietary SDKs/credentials that we deliberately do not
# vendor or fetch).
option('curl', type : 'boolean', value : false,

option('opus', type : 'boolean', value : false,
       description : 'DEAD END - Opus voice codec (was only meaningful with FMOD voice chat, which is removed; not ported)')
